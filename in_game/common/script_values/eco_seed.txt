# agricultural regime: 1 thermal (default), 2 mediterranean & arid, 3 monsoon, 4 equatorial
eco_seed_regime = {
	desc = "ECO_SV_GEOGRAPHY"
	value = 1
	if = {
		limit = { capital ?= { climate = mediterranean } }
		value = 2
	}
	if = {
		limit = { capital ?= { climate = arid } }
		value = 2
	}
	if = {
		limit = { capital ?= { climate = subtropical } }
		value = 3
	}
	if = {
		limit = { capital ?= { climate = tropical } }
		value = 3
	}
	# equatorial: tropical AND warm year-round (monthly temperature never near 20 C)
	if = {
		limit = { capital ?= { climate = tropical } var:eco_temp_mean >= 24 }
		value = 4
	}
}
