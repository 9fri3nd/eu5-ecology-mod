eco_gdd_fulfillment = {
	value = eco_gdd_rel_wheat
	min = eco_gdd_rel_rice
	min = eco_gdd_rel_maize
	min = eco_gdd_rel_millet
	min = eco_gdd_rel_potato
}

# State-driven growth factor (phenology state machine): dormant provinces do not grow,
# fallow/drought provinces stall to a trickle. Replaces the old Mediterranean calendar rows.
eco_gdd_season_factor = {
	value = 1
	if = {
		limit = { var:eco_phenology_state = 0 }
		value = 0
	}
	if = {
		limit = { var:eco_phenology_state = 2 }
		value = 0.1
	}
}
